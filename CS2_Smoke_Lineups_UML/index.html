    class ThrowType {
        <<ENUM>>
        STAND : 0
        JUMPTHROW : 1
        RUN_JUMPTHROW : 2
    }

    class Role {
        <<ENUM>>
        GUEST : 0
        USER : 1
        ADMIN : 2
    }

    class Status {
        <<ENUM>>
        PENDING : 0
        APPROVED : 1
        REJECTED : 2
    }

    class Lineup {
        -int id
        -string title
        -float startX
        -float startY
        -float endX
        -float endY
        -string videoUrl
        -string crosshairImageUrl
        -ThrowType throwType
        -string description
        -Status status
        -datetime createdAt
        +save(): void
        +updateStatus(status: Status): void
        +delete(): void
    }

    class User {
        -int id
        -string email
        -string passwordHash
        -Role role
        -datetime createdAt
        +register(): void
        +login(): string
        +logout(): void
    }

    class Map {
        -int id
        -string name
        -string radarImageUrl
        +getRadars(): List~Map~
    }

    class LineupManager {
        +getLineupsByMap(mapId: int): List~Lineup~
        +searchLineups(query: string): List~Lineup~
        +verifyLineup(lineupId: int, newStatus: Status): void
    }
